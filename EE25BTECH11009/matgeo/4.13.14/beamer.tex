\begin{frame}{Question}
The perpendicular bisector of the line segment joining $P(1,4)$ and $Q(k,3)$ has $Y$--intercept $-4$. Then a possible value of $k$ is
\begin{align}
\text{(a) }1 \quad \text{(b) }2 \quad \text{(c) }-2 \quad \text{(d) }-4
\end{align}
\end{frame}

%---------------------------------
\begin{frame}{Theoretical Solution}
Equation of perpendicular bisector:
\begin{align}
\vec{n}^{\top}\vec{x} = c \label{eq:1}
\end{align}
Direction vector of $PQ$:
\begin{align}
\vec{PQ} = \myvec{k-1 \\ -1} \label{eq:2}
\end{align}
As the perpendicular bisector has normal:
\begin{align}
\vec{n} = \myvec{k-1 \\ -1}
\end{align}
\end{frame}

%---------------------------------
\begin{frame}{Midpoint and Substitution}
Midpoint of $PQ$:
\begin{align}
\vec{M} = \myvec{\tfrac{1+k}{2} \\[4pt] \tfrac{7}{2}}
\end{align}
Y-intercept of the perpendicular bisector:
\begin{align}
c = -4 \label{eq:3}
\end{align}

Substitute ~\eqref{eq:2} and ~\eqref{eq:3} in ~\eqref{eq:1}:
\begin{align}
\myvec{k-1 & -1}\myvec{-\tfrac{1+k}{2} \\[4pt] -4-\tfrac{7}{2}}=0
\end{align}
\end{frame}

%---------------------------------
\begin{frame}{Solving for k}
\begin{align}
-(k^2 - 1) + 15 = 0
\end{align}
\begin{align}
-k^2 + 16 = 0 \quad \Rightarrow \quad k^2 = 16
\end{align}
Hence
\begin{align}
k = \pm 4
\end{align}
From the options, valid value is
\[
\boxed{-4}
\]
\end{frame}

#include <stdio.h>
void compute_bisector(double *P, double *Q, double *results)
{
    double Mx = (P[0] + Q[0]) / 2.0;
    double My = (P[1] + Q[1]) / 2.0;
    double a = Q[0] - P[0];  // Δx
    double b = Q[1] - P[1];  // Δy
\end{lstlisting}
\end{frame}

%---------------------------------
\begin{frame}[fragile]
\frametitle{C Code (Part 2)}
\begin{lstlisting}
    double c = a * Mx + b * My;
    results[0] = a;  // a
    results[1] = b;  // b
    results[2] = c;  // c

    if (b != 0) {
        results[3] = -a / b;   // slope
        results[4] = c / b;    // intercept
    } else {
        results[3] = 1e9;
        results[4] = 0;
    }
}
\end{lstlisting}

import numpy as np
import matplotlib.pyplot as plt

# Given points
P = np.array([1, 4])
k = -4     # Correct answer
Q = np.array([k, 3])

# Midpoint of PQ
M = (P + Q) / 2

# Direction vector of PQ
PQ = Q - P

# Normal vector to bisector (same as direction of PQ)
n = PQ

# Equation of bisector: n·(x - M) = 0  → a*x + b*y = c
a, b = n
c = a*M[0] + b*M[1]
\end{lstlisting}
