\begin{frame}{Question}
Yash scored 40 marks in a test, getting 3 marks for each right answer and losing 1
mark for each wrong answer. Had 4 marks been awarded for each correct answer
and 2 marks been deducted for each incorrect answer, then Yash would have scored
50 marks. How many questions were there in the test?
\end{frame}

\begin{frame}{Solution}
Let the number of right and wrong answers be $x$ and $y$ respectively.
\begin{align}
\myvec{3 & -1}\myvec{x \\ y} = 40
\end{align}
\begin{align}
\myvec{4 & -2}\myvec{x \\ y} = 50
\end{align}

These can be combined to give the matrix equation
\begin{align}
\myvec{3 & -1 \\ 4 & -2}\myvec{x \\ y} = \myvec{40 \\ 50}
\end{align}
\end{frame}

\begin{frame}{Solution}
This gives the augmented matrix
\begin{align}
\augvec{2}{1}{3 & -1 & 40 \\ 4 & -2 & 50}
\xleftrightarrow{R_2 \leftarrow R_2 - \frac{4}{3}R_1}
\augvec{2}{1}{3 & -1 & 40 \\ 0 & -\frac{2}{3} & -\frac{10}{3}}
\end{align}

This gives the following values of $x$ and $y$:
\begin{align}
\myvec{x \\ y} = \myvec{15 \\ 5}
\end{align}

Thus, the number of questions in the test is 
\begin{align}
x + y = 20
\end{align}
\end{frame}

#include <stdio.h>

int solve_2x2(double A[4], double b[2], double x[2]) {
    double det = A[0]*A[3] - A[1]*A[2]; 
    if (det == 0.0) {
        return -1; 
    }
    x[0] = (b[0]*A[3] - b[1]*A[1]) / det;
    x[1] = (A[0]*b[1] - A[2]*b[0]) / det;
    return 0; 
}

import numpy as np
import matplotlib.pyplot as plt

A = np.array([[3, -1],
              [4, -2]])
b = np.array([40, 50])
try:
    solution = np.linalg.solve(A, b)
    x_sol, y_sol = solution
    print(f"The system has a unique solution:")
    print(f"x = {x_sol}")
    print(f"y = {y_sol}")
    x_vals = np.linspace(x_sol - 40, x_sol + 40, 400)
    y1 = (3*x_vals - 40)
    y2 = (2*x_vals - 25)
    plt.figure(figsize=(8, 8))
    plt.plot(x_vals, y1, label=r'$3x - y = 40$')
    plt.plot(x_vals, y2, label=r'$4x - 2y = 50$')
